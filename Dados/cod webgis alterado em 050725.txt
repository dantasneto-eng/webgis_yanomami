        poligono_analise, how='intersection'
    )

    def calcular_area(gdf):

    area_embargos = calcular_area(entrada_embargo)
    area_tis = calcular_area(entrada_tis)

    out_image, resized_image, bounds = processar_raster(raster_subido, poligono_analise)

    st.write('Áreas em hectares por classe:')
    unique_values, counts = np.unique(out_image, return_counts=True)
    for value, count in zip(unique_values, counts):
        class_name = value_to_class.get(value, "Desconhecido")
        area_ha = (count * 900) / 10000
        st.write(f"{class_name}: {area_ha:.2f} ha")

import numpy as np
import cv2
from rasterio.io import MemoryFile
from rasterio.mask import mask
from utils import color_map

def processar_raster(raster_file, poligono):
    with MemoryFile(raster_file.getvalue()) as memfile:
        with memfile.open() as src:
            if poligono.crs != src.crs:
                poligono = poligono.to_crs(src.crs)

            out_image, out_transform = mask(src, poligono.geometry, crop=True)
            out_image = out_image[0]

            height, width = out_image.shape
            rgb_image = np.zeros((height, width, 4), dtype=np.uint8)

            for value, color in color_map.items():
                rgb_image[out_image == value] = color

            resized_image = cv2.resize(rgb_image, (width, height), interpolation=cv2.INTER_NEAREST)
            min_x, min_y = out_transform * (0, 0)
            max_x, max_y = out_transform * (width, height)
            bounds = [[min_y, min_x], [max_y, max_x]]

            return out_image, resized_image, bounds

Mapa.py

import folium
from folium.raster_layers import ImageOverlay
from streamlit_folium import st_folium

def adicionar_camada(gdf, nome, cor, mapa):
    folium.GeoJson(gdf[['geometry']], name=nome, style_function=lambda x: {
        'fillColor': cor, 'color': 'black', 'weight': 1, 'fillOpacity': 0.6
    }).add_to(mapa)

def construir_mapa(resized_image, bounds, entrada_desmat, entrada_embargo, entrada_tis):
    m = folium.Map(location=[-15, -54], zoom_start=8, tiles='Esri World Imagery')
    ImageOverlay(image=resized_image, bounds=bounds, opacity=0.7, name='Mapbiomas coleção 9').add_to(m)

    adicionar_camada(entrada_desmat, 'Área desmatada', 'red', m)
    adicionar_camada(entrada_embargo, 'Área embargada', 'orange', m)
    adicionar_camada(entrada_tis, 'Área de TIs', 'yellow', m)

    folium.LayerControl().add_to(m)
    st_folium(m, width='100%')

import plotly.express as px
import streamlit as st

def exibir_grafico(df_embargo, df_desmat, df_ti):
    col1, col2, col3, col4 = st.columns(4)
    tema = col1.selectbox('Tema', ['Desmatamento', 'Embargo', 'Terras Indígenas'])
    tipo = col2.selectbox('Tipo', ['box', 'bar', 'line', 'scatter', 'violin', 'histogram'], index=5)

    df_analisado = {
        'Embargo': df_embargo,
        'Desmatamento': df_desmat,
        'Terras Indígenas': df_ti
    }[tema]

    if not df_analisado.empty:
        x_val = col3.selectbox('Eixo X', df_analisado.columns)
        y_val = col4.selectbox('Eixo Y', df_analisado.columns)
        plot_func = getattr(px, tipo)
        st.plotly_chart(plot_func(df_analisado, x=x_val, y=y_val), use_container_width=True)
    else:
        st.info("Nenhum dado disponível para o gráfico selecionado.")

# WebGIS - Análise Espacial para Regularização Fundiária e Ambiental

Este projeto é uma aplicação WebGIS desenvolvida com Streamlit para análise espacial de áreas embargadas, desmatadas e Terras Indígenas com base em arquivos vetoriais e raster.

## 📦 Funcionalidades

- Upload de arquivos vetoriais (CAR, embargos, desmatamento, TIs)
- Processamento de raster (ex: MapBiomas)
- Cálculo de áreas por classe
- Visualização interativa com Folium
- Geração de gráficos com Plotly
